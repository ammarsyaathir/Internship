Connect your phone directly to your Google Sheet

Paste your deployed Google Apps Script URL above. When connected, every new internship, edit, or status change syncs with your Google Sheet while keeping a fast offline copy on your phone.

Don't have a Google Sheet yet? Switch to the Setup Guide tab above!
  1. Create a new Google Sheet in your Google Drive.
  2. Click Extensions → Apps Script in the top menu bar.
  3. Delete any existing code inside Code.gs and paste the complete script below (it automatically creates the header columns for you!).
  4. Click Deploy → New deployment, choose type Web app, set Execute as to Me, and set Who has access to Anyone. Copy the Web App URL and paste it in Tab 1!
Code.gs (Google Apps Script) 
function setupHeaders(sheet) {
  if (sheet.getLastRow() === 0) {
    sheet.appendRow([
      'ID', 'Timestamp', 'Company', 'Role', 'Location',
      'Place', 'Status', 'Allowance', 'Source', 'FollowUpDate', 'Notes'
    ]);
    sheet.getRange(1, 1, 1, 11).setFontWeight('bold');
  }
}

function doGet() {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  setupHeaders(sheet);
  const rows = sheet.getDataRange().getValues();
  rows.shift(); // Remove header row
  const data = rows.filter(r => r[0]).map(r => ({
    id: String(r[0]),
    timestamp: String(r[1] || ''),
    company: String(r[2] || ''),
    role: String(r[3] || ''),
    location: String(r[4] || ''),
    place: String(r[5] || 'On-site'),
    status: String(r[6] || 'Applied'),
    allowance: String(r[7] || ''),
    source: String(r[8] || ''),
    followUpDate: String(r[9] || ''),
    notes: String(r[10] || '')
  }));
  return ContentService.createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  setupHeaders(sheet);
  const body = JSON.parse(e.postData.contents);
  const action = body.action;
  const data = sheet.getDataRange().getValues();

  if (action === 'add' || action === 'edit') {
    let foundRow = -1;
    for (let i = 1; i < data.length; i++) {
      if (String(data[i][0]) === String(body.id)) {
        foundRow = i + 1;
        break;
      }
    }
    const rowValues = [
      body.id, body.timestamp, body.company, body.role, body.location,
      body.place, body.status, body.allowance, body.source, body.followUpDate || '', body.notes
    ];
    if (foundRow > 0) {
      sheet.getRange(foundRow, 1, 1, 11).setValues([rowValues]);
    } else {
      sheet.appendRow(rowValues);
    }
  } else if (action === 'updateStatus') {
    for (let i = 1; i < data.length; i++) {
      if (String(data[i][0]) === String(body.id)) {
        sheet.getRange(i + 1, 7).setValue(body.status);
        break;
      }
    }
  } else if (action === 'delete') {
    for (let i = 1; i < data.length; i++) {
      if (String(data[i][0]) === String(body.id)) {
        sheet.deleteRow(i + 1);
        break;
      }
    }
  } else if (action === 'bulkSync' && Array.isArray(body.items)) {
    body.items.forEach(item => {
      let exists = false;
      for (let i = 1; i < data.length; i++) {
        if (String(data[i][0]) === String(item.id)) { exists = true; break; }
      }
      if (!exists) {
        sheet.appendRow([
          item.id, item.timestamp, item.company, item.role, item.location,
          item.place, item.status, item.allowance, item.source, item.followUpDate || '', item.notes
        ]);
      }
    });
  }

  return ContentService.createTextOutput(JSON.stringify({ ok: true }))
    .setMimeType(ContentService.MimeType.JSON);
}
